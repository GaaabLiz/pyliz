24    @classmethod
25    def from_value(cls, value: str) -> "AiScanTool":
26        """
27        Maps a string identifier to an AiScanTool enum member.
28        Handles normalization and common aliases (e.g. 'joytag' -> TAGS).
29        """
30        normalized = value.strip().upper().replace("_", "-")
31        aliases = {
32            "TAGS": cls.TAGS,
33            "TAG": cls.TAGS,
34            "JOYTAG": cls.TAGS,
35            "TAGS-JOYTAG": cls.TAGS,
36            "TAG-JOYTAG": cls.TAGS,
37            "NSFW": cls.NSFW,
38            "OCR": cls.OCR,
39        }
40        try:
41            return aliases[normalized]
42        except KeyError as exc:

46    @classmethod
47    def normalize_many(cls, values: list[str]) -> list["AiScanTool"]:
48        """

267        """
268        Extracts text from the media using EasyOCR.
269
270        Args:
271            media: The LizMedia object to scan.
272
273        Returns:
274            AiScanResult containing extracted OCR text and boolean detection status.
275        """
276        reader = self._get_reader()
277        texts: list[str] = []
278
279        if media.is_image:
280            texts.extend(self._extract_texts(reader.readtext(str(media.path))))
281        elif media.is_video:
282            for frame in sample_video_frames(media.path, max_frames=self.video_sample_frames):
283                texts.extend(self._extract_texts(reader.readtext(frame)))
284        else:
285            return AiScanResult(ocr_text=[], ocr_detected=False)
286
287        normalized_texts = unique_preserving_order(texts)
288        return AiScanResult(ocr_text=normalized_texts, ocr_detected=bool(normalized_texts))

290    def _get_reader(self):
291        """
292        Loads and initializes EasyOCR.
293
294        Returns:
295            An easyocr.Reader instance.
296
297        Raises:
298            ImportError: if easyocr/torch dependencies are missing.
299        """
300        if self._reader is not None:
301            return self._reader
302
303        try:
304            import easyocr
305            import torch
306        except ImportError as exc:
307            raise ImportError("OCR scanning requires the optional AI dependencies. Install the 'ai' extra to enable OCR scans.") from exc
308
309        self._reader = easyocr.Reader(
310            self.languages,
311            gpu=torch.cuda.is_available(),
312            model_storage_directory=self.model_dir,
313            verbose=False,
314        )

319        """
320        Extracts raw strings from EasyOCR result tuples/lists.

324
325        Returns:
326            List of extracted strings.
327        """
328        texts: list[str] = []

332                if text:
333                    texts.append(text)

 57class JoyTagProvider(BaseAiProvider):
 58    """
 59    Provider that uses the JoyTag model to predict tags for images and video frames.
 60    Requires optional AI dependencies (torch, torchvision).
 61    """

 66        self,
 67        model_dir: str | None = None,
 68        confidence_threshold: float = 0.45,
 69        video_sample_frames: int = 5,
 70    ):
 71        """
 72        Initializes the JoyTag provider.
 73
 74        Args:
 75            model_dir: Directory to cache the model files. Defaults to PYLIZ_AI_MODELS_PATH.
 76            confidence_threshold: Minimum confidence score (0-1) to include a tag.
 77            video_sample_frames: Number of frames to sample from a video for analysis.
 78        """
 79        self.model_dir = model_dir or _DEFAULT_MODEL_DIR
 80        self.confidence_threshold = confidence_threshold
 81        self.video_sample_frames = video_sample_frames
 82        self._model = None
 83        self._tag_list = None
 84        self._device = None
 85        self._tvf = None
 86        self._torch = None
 87
 88    def scan(self, media: "LizMedia") -> AiScanResult:
 89        """
 90        Analyzes the media and predicts relevant tags using JoyTag.
 91
 92        Args:
 93            media: The media object to scan (supports Image and Video).
 94
 95        Returns:
 96            AiScanResult with predicted tags. Returns empty list if media type is unsupported.
 97        """
 98        if media.extension.lstrip(".") not in _IMAGE_EXTENSIONS | _VIDEO_EXTENSIONS:
 99            return AiScanResult(tags=[])
100
101        model, tag_list, device, tvf, torch = self._get_runtime()
102        predicted_tags: list[str] = []
103
104        if media.is_image:
105            with Image.open(media.path) as image_file:
106                image = image_file.convert("RGB")
107                predicted_tags.extend(self._predict_from_image(image, model, tag_list, device, tvf, torch))
108        elif media.is_video:
109            for frame in sample_video_frames(media.path, max_frames=self.video_sample_frames):
110                image = Image.fromarray(frame)
111                predicted_tags.extend(self._predict_from_image(image, model, tag_list, device, tvf, torch))
112        else:
113            return AiScanResult(tags=[])
114
115        return AiScanResult(tags=unique_preserving_order(predicted_tags))
116
117    def _get_runtime(self):
118        """
119        Loads and initializes Torch, the JoyTag model, and its dependencies.
120        Downloads the model from Hugging Face Hub if not cached.
121
122        Returns:
123            A tuple of (model, tag_list, device, torchvision_functional, torch_module).
124

180        self,
181        image,
182        model,
183        tag_list,
184        device,
185        tvf,
186        torch,
187    ) -> list[str]:
188        """
189        Runs model inference on a single PIL Image.
190
191        Args:
192            image: PIL Image object.
193            model: Loaded JoyTag model.
194            tag_list: List of supported tag names.
195            device: Torch device.
196            tvf: torchvision.transforms.functional module.
197            torch: torch module.
198
199        Returns:
200            List of tags exceeding the confidence threshold.
201        """
202        resample_mode = getattr(getattr(Image, "Resampling", Image), "BILINEAR")
203
204        image = image.resize((448, 448), resample=resample_mode)
205        image_tensor = tvf.to_tensor(image)
206        image_tensor = tvf.normalize(
207            image_tensor,
208            [0.48145466, 0.4578275, 0.40821073],
209            [0.26862954, 0.26130258, 0.27577711],
210        )
211        image_tensor = image_tensor.unsqueeze(0).to(device)
212
213        start_time = time.time()
214        with torch.no_grad():
215            batch = {"image": image_tensor}
216            output = model(batch)
217
218            if isinstance(output, dict):
219                preds = output.get("tags")
220                if preds is None:
221                    for value in output.values():
222                        if hasattr(value, "shape") and value.shape[-1] == len(tag_list):
223                            preds = value
224                            break
225                    if preds is None:
226                        preds = list(output.values())[0]
227            elif isinstance(output, (tuple, list)):
228                preds = output[0]
229            else:
230                preds = output
231
232            if preds.dim() > 1:
233                preds = preds.squeeze(0)
234            preds = torch.sigmoid(preds)
235
236        logger.debug(f"JoyTag inference completed in {round(time.time() - start_time, 2)}s for {len(tag_list)} tags.")
237        return [tag_list[i] for i, prediction in enumerate(preds) if prediction > self.confidence_threshold]
