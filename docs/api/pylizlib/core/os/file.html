25def get_file_c_date(path_to_file) -> datetime:
26    """Return a file's creation date when available, otherwise its modification date.
27
28    On Windows, use the creation timestamp. On platforms that do not expose a
29    birth time, use the last modification timestamp.
30
31    Args:
32        path_to_file (str | Path): Existing file whose timestamp should be read.
33
34    Returns:
35        datetime: Local datetime corresponding to the selected filesystem timestamp.
36    """
37    if platform.system() == "Windows":
38        timestamp = os.path.getctime(path_to_file)
39    else:
40        stat = os.stat(path_to_file)
41        try:
42            timestamp = stat.st_birthtime
43        except AttributeError:
44            timestamp = stat.st_mtime
45
46    return datetime.fromtimestamp(timestamp)

def write_json_to_file(path, filename, content): 
 9def write_json_to_file(path, filename, content):
10    """Serialize JSON and atomically replace a file in its parent directory.
11
12    Args:
13        path (str | Path): Parent directory for the output file; created if missing.
14        filename (str): Output file name relative to ``path``.
15        content (object): JSON-serializable value to write.
16
17    Returns:
18        None: The serialized file has replaced the destination.
19
20    Raises:
21        TypeError: The value cannot be serialized as JSON.
22        OSError: Staging or replacing the output file fails.
23    """
24    if not os.path.exists(path):
25        os.makedirs(path)
26
27    # Serialize first so invalid values cannot truncate an existing file.
28    serialized = json.dumps(content, indent=4)
29    file_path = os.path.join(path, filename)
30    temporary_path = None
31    try:
32        with tempfile.NamedTemporaryFile(
33            mode="w",
34            encoding="utf-8",
35            dir=path,
36            prefix=".json-",
37            delete=False,
38        ) as json_file:
39            temporary_path = json_file.name
40            json_file.write(serialized)
41            json_file.flush()
42            os.fsync(json_file.fileno())
43        os.replace(temporary_path, file_path)
44        temporary_path = None
45    finally:
46        if temporary_path is not None:
47            try:
48                os.unlink(temporary_path)
49            except FileNotFoundError:
50                pass
