487    def match_with_list(self, path_str_list: list[str]):
488        """Compare the loaded relative paths with another path list.
489
490        Args:
491            path_str_list (list[str]): Relative path names to compare against.
492
493        Returns:
494            tuple[int, float]: Intersection count and intersection-over-union percentage.
495        """
496        set1, set2 = set(self.working_path_items_rel), set(path_str_list)
497        intersection = len(set1 & set2)
498        union = len(set1 | set2)
499        perc = (intersection / union) * 100 if union > 0 else 100
500        return intersection, perc

507
508        Returns:
509            tuple[int, float]: Intersection count and intersection-over-union percentage.
510        """
511        with open(file_path, "r") as file:
512            return self.match_with_list([line.strip() for line in file.readlines()])

def duplicate_directory( src_dir: pathlib.Path, dest_dir: pathlib.Path | None = None, copy_suffix: str = '_copy', message_callback=None) -> pathlib.Path: 
405def duplicate_directory(src_dir: Path, dest_dir: Path | None = None, copy_suffix: str = "_copy", message_callback=None) -> Path:
406    """Copy a directory tree and return the new destination path.
407
408    If ``dest_dir`` is omitted, the copy is a sibling named by appending
409    ``copy_suffix`` to the source directory name.
410
411    Args:
412        src_dir (Path): Existing directory tree to copy.
413        dest_dir (Path | None): Optional destination path for the copy.
414        copy_suffix (str): Suffix used for a default sibling destination.
415        message_callback (Callable[[str], object] | None): Optional callback receiving each copied file name.
416
417    Returns:
418        Path: Path to the completed copy.
419
420    Raises:
421        ValueError: The source is not a directory or the destination is inside the source.
422        FileExistsError: The destination already exists.
423    """
424    if not src_dir.is_dir():
425        raise ValueError(f"{src_dir!r} non è una directory valida")
426
427    # Se non specificato, costruisce dest_dir affiancata a src_dir
428    if dest_dir is None:
429        dest_dir = src_dir.with_name(src_dir.name + copy_suffix)
430
431    resolved_src = src_dir.resolve(strict=True)
432    resolved_dest = dest_dir.resolve(strict=False)
433    if resolved_dest == resolved_src or resolved_src in resolved_dest.parents:
434        raise ValueError("The destination directory cannot be the source or one of its descendants")
435
436    # Assicura che la destinazione non esista già
437    if dest_dir.exists():
438        raise FileExistsError(f"{dest_dir!r} esiste già")
439
440    def copy_with_cb(src, dst, *, follow_symlinks=True):
441        """Copy one file entry and report it to the optional progress callback.
442
443        Args:
444            src (str | Path): Source file path.
445            dst (str | Path): Destination file path.
446            follow_symlinks (bool): Whether ``copy2`` follows a source symlink.
447
448        Returns:
449            None: The file metadata and contents have been copied.
450        """
451        if message_callback:
452            message_callback(f"Copia in corso: {Path(src).name}")
453        shutil.copy2(src, dst, follow_symlinks=follow_symlinks)
454
455    # Copia ricorsivamente la directory
456    shutil.copytree(src_dir, dest_dir, copy_function=copy_with_cb)
457    return dest_dir
