def assert_destination_outside_source(source: pathlib.Path | str, destination: pathlib.Path | str) -> None: 
 7def assert_destination_outside_source(source: Path | str, destination: Path | str) -> None:
 8    """Reject recursive tree copies, resolving both paths through aliases.
 9
10    Args:
11        source (Path | str): Source tree, resolved without requiring every entry to exist.
12        destination (Path | str): Proposed copy target, resolved using the same policy.
13
14    Returns:
15        None: If destination is neither source nor structurally inside source.
16
17    Raises:
18        ValueError: If destination resolves to source or one of its descendants.
19        OSError: If the filesystem prevents resolving either path.
20
21    Notes:
22        This check does not reserve paths against concurrent filesystem changes.
23    """
24    source = Path(source)
25    destination = Path(destination)
26    source_resolved = source.resolve(strict=False)
27    destination_resolved = destination.resolve(strict=False)
28    if destination_resolved.is_relative_to(source_resolved):
29        raise ValueError(f"Copy destination '{destination}' is inside source directory '{source}'.")

def normalize_rooted_path( value: str, *, allow_relative: bool = True, path_label: str = 'Path', drive_type: Callable[[str], int] | None = None, unc_root: Callable[[str], str] | None = None) -> str: 
51def normalize_rooted_path(value: str, *, allow_relative: bool = True, path_label: str = "Path", drive_type: Callable[[str], int] | None = None, unc_root: Callable[[str], str] | None = None) -> str:
52    """Return an absolute, lexically normalized path without following symlinks.
53
54    Args:
55        value (str): Non-empty POSIX or Windows path without NUL characters.
56        allow_relative (bool): Whether to anchor a relative path to CWD.
57        path_label (str): Subject used in validation messages; defaults to Path.
58        drive_type (Callable[[str], int] | None): Optional drive-letter type lookup. Omitted uses the native
59            Windows adapter, called only on Windows.
60        unc_root (Callable[[str], str] | None): Optional mapped-drive UNC lookup.
61
62    Returns:
63        str: Rooted spelling with forward slashes; links and parent components remain unresolved. Windows remote
64            drives are converted to UNC.
65
66    Raises:
67        TypeError: If value is not a string.
68        ValueError: If the input is invalid or a required UNC lookup fails.
69        OSError: If CWD or a native drive operation cannot be read.
70
71    Notes:
72        The path need not exist. This operation does not establish containment.
73    """
74    if not isinstance(value, str):
75        raise TypeError(f"{path_label} must be a string.")
76    if not value or "\x00" in value:
77        raise ValueError(f"{path_label} must be a non-empty path without NUL characters.")
78
79    # Recognize foreign Windows syntax without interpreting it as a local relative path.
80    windows_path = PureWindowsPath(value)
81    if windows_path.drive and not windows_path.root:
82        raise ValueError(f"Drive-relative paths are not supported: {value!r}.")
83
84    if windows_path.is_absolute():
85        if os.name == "nt" and not is_unc_path(windows_path) and (drive_type or get_drive_type)(windows_path.drive[0]) == 4:
86            return normalize_mapped_drive(windows_path, unc_root=unc_root)
87        return windows_path.as_posix()
88
89    # POSIX roots remain lexical and do not cause filesystem link resolution.
90    posix_path = PurePosixPath(value)
91    if posix_path.is_absolute():
92        return posix_path.as_posix()
93
94    if not allow_relative:
95        raise ValueError(f"{path_label} must be absolute; relative path {value!r} cannot be resolved safely.")
96
97    # Anchor without resolving links or collapsing ``..`` across a symlink.
98    return (Path.cwd() / Path(value)).as_posix()

def same_filesystem_entry(first: str | pathlib.Path, second: str | pathlib.Path) -> bool: 
 7def same_filesystem_entry(first: str | Path, second: str | Path) -> bool:
 8    """Return whether two paths name the same existing filesystem entry.
 9
10    Path equality handles the platform's lexical rules. ``samefile`` adds the
11    filesystem's own rules for aliases such as symbolic links and
12    case-insensitive names. Missing entries are not equivalent unless their
13    paths compare equal lexically.
14
15    Args:
16        first (str | Path): First native-platform path; it need not exist.
17        second (str | Path): Path to compare with first.
18
19    Returns:
20        bool: True for equal lexical paths or aliases of one existing entry. False for unequal paths when either
21            entry is missing.
22
23    Raises:
24        ValueError: If an OS error other than a missing entry prevents comparison. The original OSError remains
25            available as the exception cause.
26    """
27    first_path = Path(first)
28    second_path = Path(second)
29    if first_path == second_path:
30        return True
31    try:
32        return first_path.samefile(second_path)
33    except FileNotFoundError:
34        return False
35    except OSError as error:
36        raise ValueError(f"Cannot verify whether paths '{first_path}' and '{second_path}' are equivalent: {error}") from error
