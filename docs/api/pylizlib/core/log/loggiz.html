 23    @staticmethod
 24    def _reset_handlers(target_logger: logging.Logger) -> None:
 25        """Close and remove all handlers from a logger."""
 26
 27        for handler in list(target_logger.handlers):
 28            handler.flush()
 29            handler.close()
 30            target_logger.removeHandler(handler)

 32    @staticmethod
 33    def setup(
 34        app_name="app",
 35        setup_console=True,
 36        setup_file=True,
 37        console_level=default_console_log_level,
 38        console_format=default_console_log_format,
 39        console_ansi=False,
 40        file_level=default_file_log_level,
 41        file_format=default_file_log_format,
 42        file_log_base_path="",
 43        file_log_folder_name="logs",
 44        file_log_name="latest.log",
 45        use_app_timestamp_template=True,
 46    ):
 47        """Configure console and/or file logging handlers."""
 48
 49        if setup_file:
 50            if use_app_timestamp_template:
 51                file_log_name = Loggiz.create_timestamp_log_file_name(app_name)
 52            else:
 53                file_log_name = file_log_name
 54            log_folder_dir = os.path.join(file_log_base_path, file_log_folder_name)
 55            file_full_path = os.path.join(log_folder_dir, file_log_name)
 56            if not os.path.exists(log_folder_dir):
 57                os.makedirs(log_folder_dir)
 58            config.set_file_log_path(file_full_path)
 59            config.set_logger_file_enabled(True)
 60            Loggiz.setup_file(
 61                level=file_level,
 62                log_format=file_format,
 63                file_path=file_full_path,
 64            )
 65
 66        if setup_console:
 67            Loggiz.setup_console(
 68                level=console_level,
 69                log_format=console_format,
 70            )
 71            config.set_console_ansi(console_ansi)
 72            config.set_logger_console_enabled(True)

195    @staticmethod
196    def get_tag_string(tag, ansify=False):
197        """Build a tag prefix string, optionally with ANSI style."""
198
199        if ansify:
200            return f"{Colors.LIGHT_WHITE}{config.tag_char_start}{tag}{config.tag_char_end}{Colors.END}"
201        else:
202            return f"{config.tag_char_start}{tag}{config.tag_char_end}"
203
204    @staticmethod
205    def print_separator(sep, count):
206        """Print a separator line to the console."""
207
208        string = f"{sep}" * count
209        if config.enable_console_ansi:
210            print(Colors.LIGHT_GRAY + string + Colors.END)
211        else:
212            print(string)

222        else:
