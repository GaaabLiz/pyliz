122    def execute_tasks(self):
123        for task in self.tasks:
124            if self._is_cancelled or task.is_cancelled:
125                task.update_task_status(OperationStatus.Cancelled)
126                self.signals.task_finished.emit(task.name)
127                continue
128            try:
129                task.on_progress_changed = self.on_task_progress_update
130                self.signals.task_start.emit(task.name)
131                task.update_task_status(OperationStatus.InProgress)
132                logger.debug("Executing task: %s", task.name)
133                self.current_task = task
134                result = task.execute()
135                task.result = result
136                if self._is_cancelled or task.is_cancelled:
137                    task.update_task_status(OperationStatus.Cancelled)
138                else:
139                    task.update_task_progress(100)
140                    task.update_task_status(OperationStatus.Completed)
141            except Exception as e:
142                task.update_task_status(OperationStatus.Failed)
143                logger.error("Error in task %s: %s", task.name, e)
144                self.signals.task_failed.emit(task.name, str(e))
145                self.task_errors.append(f"Task {task.name} failed: {e}")
146                if task.abort_all_on_error:
147                    raise RuntimeError(f"Task {task.name} failed: {e}")
148            finally:
149                self.signals.task_finished.emit(task.name)
150                self.current_task = None
151                sleep(self.info.delay_each_task)
152
153    def execute(self):
154        try:
155            self.set_operation_started()
156            self.update_op_status(OperationStatus.InProgress)
157            self.execute_tasks()
158            if self.task_errors:
159                self.error = "; ".join(self.task_errors)
160                self.update_op_status(OperationStatus.Failed)
161            elif self._is_cancelled or any(task.is_cancelled for task in self.tasks):
162                self.update_op_status(OperationStatus.Cancelled)
163            else:
164                self.update_op_status(OperationStatus.Completed)
165        except Exception as e:
166            self.update_op_status(OperationStatus.Failed)
167            self.signals.op_failed.emit(self.id, str(e))
168            self.error = str(e)

203    def update_op_status(self, status: OperationStatus):
204        logger.debug("Updating operation status: %s", status)
205        self.status = status

210    def update_op_progress(self, progress: int):
211        logger.debug("Updating operation progress: %s", progress)
212        self.progress = progress
213        self.__update_times()
214        self.signals.op_update_progress.emit(self.id, progress)
215        if self.op_progress_update_callback:
216            self.op_progress_update_callback(self.id, self.progress)
217        self.signals.op_update.emit(self)

219    def __update_times(self):
220        if self.running:
221            self.time_elapsed = time.perf_counter() - self.time_started
222            if self.progress > 0:
223                self.time_estimated_total = self.time_elapsed / (self.progress / 100)
224                self.time_estimated_remaining = max(0, self.time_estimated_total - self.time_elapsed)
225                self.signals.op_eta_update.emit(self.id, self.get_eta_formatted())
226        else:
227            if self.time_started and self.time_finished:
228                self.time_elapsed = self.time_finished - self.time_started

236    def set_operation_started(self):
237        logger.info("Starting operation %s", self.id)
238        self.running = True
239        self.signals.op_start.emit()
240        self.time_started = time.perf_counter()
241        self.__update_times()
242
243    def set_operation_finished(self):
244        logger.info("Finishing operation %s", self.id)
245        self.running = False
246        self.time_finished = time.perf_counter()
247        self.__update_times()
248        if self.finished_callback:
249            self.finished_callback()
250        self.signals.op_finished.emit(self)

252    def get_elapsed_formatted(self) -> str:
253        """Restituisce il tempo trascorso nel formato mm:ss"""
254        minutes = int(self.time_elapsed) // 60
255        seconds = int(self.time_elapsed) % 60
256        return f"{minutes:02.0f}:{seconds:02.0f}"
257
258    def get_eta_formatted(self) -> str:
259        """
260        Restituisce il tempo stimato rimanente nel formato mm:ss.
261        """
262        if self.progress <= 0:
263            return "--:--"
264
265        minutes = int(self.time_estimated_remaining) // 60
266        seconds = int(self.time_estimated_remaining) % 60
267        return f"{minutes:02}:{seconds:02}"
